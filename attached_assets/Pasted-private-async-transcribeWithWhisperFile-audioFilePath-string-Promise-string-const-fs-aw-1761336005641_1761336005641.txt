private async transcribeWithWhisperFile(audioFilePath: string): Promise<string> {
    const fs = await import('fs');

    try {
      console.log(`🎤 ===== INICIANDO TRANSCRIÇÃO COM WHISPER =====`);
      console.log(`🎤 Arquivo: ${audioFilePath}`);

      // Verificar se o arquivo existe
      if (!fs.existsSync(audioFilePath)) {
        throw new Error('Arquivo de áudio não encontrado');
      }

      // Criar FormData com o arquivo
      const formData = new FormData();

      // Ler o arquivo como buffer
      const audioBuffer = fs.readFileSync(audioFilePath);

      // Criar Blob do arquivo
      const audioBlob = new Blob([audioBuffer], { type: 'audio/ogg' });

      // Adicionar arquivo ao FormData
      formData.append('file', audioBlob, 'audio.ogg');
      formData.append('model', 'whisper-1');
      formData.append('language', 'pt');
      formData.append('response_format', 'json');

      console.log(`🎤 📤 Enviando arquivo para OpenAI Whisper API...`);

      const response = await fetch('https://api.openai.com/v1/audio/transcriptions', {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${this.openaiApiKey}`,
        },
        body: formData,
      });

      console.log(`🎤 📥 Resposta da API: ${response.status} ${response.statusText}`);

      if (!response.ok) {
        const errorText = await response.text();
        console.error(`❌ Erro na API OpenAI:`, response.status, errorText);
        throw new Error(`API Error ${response.status}: ${errorText}`);
      }

      const data = await response.json();
      console.log(`🎤 📋 Dados recebidos do Whisper:`, data);

      const transcription = data.text?.trim() || '';

      if (!transcription) {
        throw new Error('Transcrição vazia recebida da API');
      }

      console.log(`🎤 ✅ TRANSCRIÇÃO BEM-SUCEDIDA: "${transcription}"`);
      return transcription;
    } catch (error: any) {
      console.error('❌ ERRO NA TRANSCRIÇÃO:', {
        message: error.message,
        stack: error.stack,
        name: error.name
      });
      return "Oi! Recebi seu áudio mas tive um probleminha pra entender. Pode repetir ou me mandar por texto? Tô aqui pra ajudar! 😊";
    }
  }

  /**
   * Salva o buffer de áudio como arquivo .ogg temporário
   */
  private async saveAudioAsOggFile(audioBuffer: Buffer): Promise<string | null> {
    const fs = await import('fs');
    const path = await import('path');

    try {
      // Criar nome único para o arquivo temporário
      const timestamp = Date.now();
      const randomId = Math.random().toString(36).substring(7);
      const fileName = `audio_${timestamp}_${randomId}.ogg`;
      const filePath = path.join(process.cwd(), 'temp', fileName);

      // Criar diretório temp se não existir
      const tempDir = path.join(process.cwd(), 'temp');
      if (!fs.existsSync(tempDir)) {
        fs.mkdirSync(tempDir, { recursive: true });
      }

      // Salvar arquivo
      fs.writeFileSync(filePath, audioBuffer);

      console.log(`🎤 💾 Áudio salvo em: ${filePath}`);
      console.log(`🎤 📊 Tamanho do arquivo: ${audioBuffer.length} bytes`);

      return filePath;
    } catch (error: any) {
      console.error('❌ Erro ao salvar arquivo de áudio:', error);
      return null;
    }
  }

  /**
   * Remove arquivo temporário
   */
  private async cleanupTempFile(filePath: string): Promise<void> {
    const fs = await import('fs');

    try {
      if (fs.existsSync(filePath)) {
        fs.unlinkSync(filePath);
        console.log(`🎤 🗑️ Arquivo temporário removido: ${filePath}`);
      }
    } catch (error: any) {
      console.error('❌ Erro ao remover arquivo temporário:', error);
    }
  }

  /**
   * Valida se uma string é base64 válido
   */
  private isValidBase64(str: string): boolean {
    try {
      // Limpar quebras de linha e espaços
      const cleanBase64 = str.replace(/\s/g, '');

      // Tentar criar um buffer do base64
      Buffer.from(cleanBase64, 'base64');

      // Verificar se tem caracteres válidos de base64
      const base64Regex = /^[A-Za-z0-9+/]*={0,2}$/;
      return base64Regex.test(cleanBase64) && cleanBase64.length > 100; // Áudio deve ter pelo menos 100 chars
    } catch (error) {
      return false;
    }
  }

  /**
   * Limpa e valida base64
   */
  private cleanBase64(base64: string): string {
    return base64.replace(/\s/g, ''); // Remove quebras de linha e espaços
  }
